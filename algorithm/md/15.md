Union Find 模版

```kotlin

class UnionFind(min: Int, max: Int) {

    private val father: IntArray = IntArray(max + 1)
    private val sizes: IntArray = IntArray(max + 1)

    init {
        for (i in min..max) {
            father[i] = i
            sizes[i] = 1
        }
    }

    fun union(a: Int, b: Int) {
        val rootA = find(a)
        val rootB = find(b)
        if (rootA != rootB) {
            father[rootA] = rootB
            sizes[rootB] += sizes[rootA]
        }
    }

    fun isInSameSet(a: Int, b: Int): Boolean {
        return find(a) == find(b)
    }

    fun find(x: Int): Int {
        if (father[x] == x) {
            return x
        }
        father[x] = find(father[x])
        return father[x]
    }

    fun sizeOfSetContains(x: Int): Int {
        return sizes[find(x)]
    }
}
```

第二号版本， 可以查询有几个set 和set大小
```kotlin

private class UnionFind(min: Int, max: Int) {
    private val father: IntArray = IntArray(max + 1)
    private val size: IntArray = IntArray(max + 1)
    var countSets: Int = max - min + 1
        private set

    init {
        for (i in min..max) {
            father[i] = i
            size[i] = 1
        }
    }

    fun find(x: Int): Int {
        if (father[x] != x) {
            father[x] = find(father[x])
        }
        return father[x]
    }

    fun union(a: Int, b: Int) {
        val rootA = find(a)
        val rootB = find(b)

        if (rootA != rootB) {
            father[rootA] = rootB
            size[rootB] += size[rootA]
            countSets--
        }
    }

    fun isInSameSet(a: Int, b: Int): Boolean {
        return find(a) == find(b)
    }

    fun sizeOfSetContains(x: Int): Int {
        return size[find(x)]
    }
}
```
