```kotlin
class UnionFind(min: Int, max: Int) {

    private val father: IntArray = IntArray(max + 1)
    private val sizes: IntArray = IntArray(max + 1)

    init {
        for (i in min..max) {
            father[i] = i
            sizes[i] = 1
        }
    }

    fun union(a: Int, b: Int) {
        val rootA = find(a)
        val rootB = find(b)
        if (rootA != rootB) {
            father[rootA] = rootB
            sizes[rootB] += sizes[rootA]
        }
    }

    fun isInSameSet(a: Int, b: Int): Boolean {
        return find(a) == find(b)
    }

    fun find(x: Int): Int {
        if (father[x] == x) {
            return x
        }
        father[x] = find(father[x])
        return father[x]
    }

    fun sizeOfSetContains(x: Int): Int {
        return sizes[find(x)]
    }
}
```

```kotlin
class Solution {

        val n = isConnected.size

        isConnected.forEachIndexed{ i, row->

            }

        }
```

```kotlin
class Solution {
    fun equationsPossible(equations: Array<String>): Boolean {
        val uf = UnionFind(26)
        
        for (eq in equations) {
            if (eq.contains("==")) {
                val first = eq[0] - 'a'
                val second = eq[3] - 'a'
                uf.union(first, second)
            }
        }
        
        for (eq in equations) {
            if (eq.contains("!=")) {
                val first = eq[0] - 'a'
                val second = eq[3] - 'a'
                if (uf.inTheSameSet(first, second)) return false
            }
        }
        
        return true
    }
}
```

[Number of Islands II](https://leetcode.com/problems/number-of-islands-ii/description/)
```kotlin
class Solution {
    private val directionX = intArrayOf(0, 0, 1, -1)
    private val directionY = intArrayOf(1, -1, 0, 0)

    fun numIslands2(m: Int, n: Int, positions: Array<IntArray>): List<Int> {
        val res = mutableListOf<Int>()
        val uf = UnionFind(m * n)
        val grid = Array(m) { IntArray(n) { 0 } }

        for (position in positions) {
            val i = position[0]
            val j = position[1]
            val index = i * n + j

            if (grid[i][j] == 1) {
                res.add(uf.countSets)
                continue
            }

            grid[i][j] = 1
            uf.add(index)

            for (k in 0 until 4) {
                val nextX = i + directionX[k]
                val nextY = j + directionY[k]

                if (nextX in 0 until m && nextY in 0 until n && grid[nextX][nextY] == 1) {
                    val neighborIndex = nextX * n + nextY
                    if (uf.contains(neighborIndex)) {
                        uf.union(neighborIndex, index)
                    }
                }
            }

            res.add(uf.countSets)
        }

        return res
    }

    class UnionFind(private val n: Int) {
        var countSets = 0
        private val parent = IntArray(n) { -1 }
        private val rank = IntArray(n) { 1 }

        fun find(x: Int): Int {
            if (parent[x] != x) {
                parent[x] = find(parent[x]) // Path compression
            }
            return parent[x]
        }

        fun add(newElem: Int) {
            if (!contains(newElem)) {
                parent[newElem] = newElem
                countSets++
            }
        }

        fun contains(x: Int): Boolean {
            return parent[x] != -1
        }

        fun union(a: Int, b: Int) {
            val rootA = find(a)
            val rootB = find(b)
            if (rootA == rootB) return

            // Union by rank

            } else {
                parent[rootB] = rootA
            }

            countSets-- // Decrease count of sets as two sets are merged
        }
    }
}
```

```
class UnionFind<T> {

    private val father: MutableMap<T, T> = HashMap()
    private val size: MutableMap<T, Int> = HashMap()
    private var countSets: Int = 0

    fun getCountSets(): Int {
        return this.countSets
    }

    fun contains(x: T): Boolean {
        return this.father.containsKey(x)
    }

    fun add(newElem: T) {
        if (!this.contains(newElem)) {
            this.father[newElem] = newElem
            this.size[newElem] = 1
            this.countSets++
        }
    }

    fun find(x: T): T? {
        if (!this.father.containsKey(x)) {
            return null
        }

        if (this.father[x] == x) {
            return x
        }

        val ret = this.find(this.father[x]!!)
        this.father[x] = ret
        return ret
    }

    fun union(a: T, b: T) {
        val rootA = this.find(a)
        val rootB = this.find(b)

        if (rootA != null && rootB != null && rootA != rootB) {
            this.father[rootA] = rootB
            if (this.size.containsKey(rootB) && this.size.containsKey(rootA)) {
                this.size[rootB] = this.size[rootB]!! + this.size[rootA]!!
            }
            this.countSets--
        }
    }

    fun isInSameSet(a: T, b: T): Boolean {
        val rootA = this.find(a)
        val rootB = this.find(b)

        return rootA == rootB
    }

    fun sizeOfSetContains(x: T): Int {
        val rootX = this.find(x)

        if (rootX != null && this.size.containsKey(rootX)) {
            return this.size[rootX]!!
        }

        return 0
    }
}
```

```
class UnionFind<T> {

    private val father = mutableMapOf<T, T>()
    private val size = mutableMapOf<T, Int>()
    private var countSets = 0

    fun getCountSets(): Int {
        return countSets
    }

    fun contains(x: T): Boolean {
        return father.containsKey(x)
    }

    fun add(newElem: T) {
        if (!contains(newElem)) {
            father[newElem] = newElem
            size[newElem] = 1
            countSets++
        }
    }

    fun find(x: T): T? {
        if (!father.containsKey(x)) {
            return null
        }

        if (father[x] == x) {
            return x
        }

        val ret = find(father[x]!!)
        father[x] = ret!!
        return ret
    }

    fun union(a: T, b: T) {
        val rootA = find(a)
        val rootB = find(b)

        if (rootA != null && rootB != null && rootA != rootB) {
            father[rootA] = rootB
            if (size.containsKey(rootB) && size.containsKey(rootA)) {
                size[rootB] = size[rootB]!! + size[rootA]!!
            }
            countSets--
        }
    }

    fun isInSameSet(a: T, b: T): Boolean {
        val rootA = find(a)
        val rootB = find(b)

        return rootA == rootB
    }

    fun sizeOfSetContains(x: T): Int {
        val rootX = find(x)

        if (rootX != null && size.containsKey(rootX)) {
            return size[rootX]!!
        }

        return 0
    }

    fun getAllDisjointSets(): Map<T, Set<T>> {
        val rets = mutableMapOf<T, MutableSet<T>>()

        val roots = getRoots()

        for (root in roots) {
            rets[root] = summon(root)
        }

        return rets
    }

    private fun summon(root: T): MutableSet<T> {
        val rets = mutableSetOf<T>()

        val keys = father.keys

        for (key in keys) {
            if (find(key) == root) {
                rets.add(key)
            }
        }

        return rets
    }

    private fun getRoots(): Set<T> {
        val rets = mutableSetOf<T>()

        val keys = father.keys
        for (key in keys) {
            if (father[key] == key) {
                rets.add(key)
            }
        }

        return rets
    }
}
```
