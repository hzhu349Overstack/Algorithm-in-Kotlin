```kotlin
class UnionFind(min: Int, max: Int) {

    private val father: IntArray = IntArray(max + 1)
    private val sizes: IntArray = IntArray(max + 1)

    init {
        for (i in min..max) {
            father[i] = i
            sizes[i] = 1
        }
    }

    fun union(a: Int, b: Int) {
        val rootA = find(a)
        val rootB = find(b)
        if (rootA != rootB) {
            father[rootA] = rootB
            sizes[rootB] += sizes[rootA]
        }
    }

    fun isInSameSet(a: Int, b: Int): Boolean {
        return find(a) == find(b)
    }

    fun find(x: Int): Int {
        if (father[x] == x) {
            return x
        }
        father[x] = find(father[x])
        return father[x]
    }

    fun sizeOfSetContains(x: Int): Int {
        return sizes[find(x)]
    }
}
```

[547. Number of Provinces](https://leetcode.com/problems/number-of-provinces/)

```kotlin
class Solution {
    fun findCircleNum(isConnected: Array<IntArray>): Int {
        val n = isConnected.size

        val uf = UnionFind(n)

        isConnected.forEachIndexed{ i, row-> 

            row.forEachIndexed{ j, value -> 
                if( i != j && value == 1)
                    uf.union(i, j)
            }
        }

        return uf.numOfSets
    }

    class UnionFind(private val n: Int){
        var numOfSets = n

        private val parent = IntArray(n){it}
        private val rank = IntArray(n)

        fun find(x: Int): Int{
            if( parent[x] != x)
                parent[x] = find(parent[x])

            return parent[x]
        }

        fun union(a: Int, b: Int){
            val rootA = find(a)
            val rootB = find(b)
            if(rootA == rootB) return 

            if( rank[rootA] < rank[rootB])
                parent[rootA] = rootB
            else if( rank[rootA] == rank[rootB]){
                parent[rootA] = rootB 
                rank[rootB]++
            } else 
                parent[rootB] = rootA

            numOfSets--
        }
    }
}
```

[2782. Number of Unique Categories](https://leetcode.com/problems/number-of-unique-categories/description/)


```kotlin
/**
 * Definition for a category handler.
 * class CategoryHandler(categories: IntArray) {
 *     fun haveSameCategory(a: Int, b: Int): Boolean
 * }
 */
class Solution {
    fun numberOfCategories(n: Int, categoryHandler: CategoryHandler): Int {
        val uf = UnionFind(n)
        for (i in 0 until n) {
            for (j in i + 1 until n) {
                if (categoryHandler.haveSameCategory(i, j)) {
                    uf.union(i, j)
                }
            }
        }

        return uf.numOfSets
    }

    class UnionFind(private val n: Int) {
        private val parent = IntArray(n) { it }
        private val rank = IntArray(n)
        var numOfSets = n

        private fun find(x: Int): Int {
            if (parent[x] != x) {
                parent[x] = find(parent[x])
            }
            return parent[x]
        }

        fun union(a: Int, b: Int) {
            val rootA = find(a)
            val rootB = find(b)
            if (rootA == rootB) return

            if (rank[rootA] < rank[rootB]) {
                parent[rootA] = rootB
            } else if (rank[rootA] == rank[rootB]) {
                parent[rootA] = rootB
                rank[rootB]++
            } else {
                parent[rootB] = rootA
            }

            numOfSets--
        }
    }
}
```

UnionFind 第四号模版针对一些union的状态一直在变化，如岛屿II一开始是个空集，然后不停的有新的陆地产生，需要进行union操作

```kotlin
class UnionFind(min: Int, max: Int, isEmpty: Boolean) {
    private val size: IntArray = IntArray(max + 1)
    private val father: IntArray = IntArray(max + 1)
    var countSets: Int = if (isEmpty) 0 else max - min + 1
        private set
    private val impossibleValue: Int = if (isEmpty) min - 1 else 0

    init {
        for (i in min..max) {
            if (isEmpty) {
                father[i] = impossibleValue
                size[i] = 0
            } else {
                father[i] = i
                size[i] = 1
            }
        }
    }

    fun add(newElem: Int) {
        if (!contains(newElem)) {
            father[newElem] = newElem
            size[newElem] = 1
            countSets++
        }
    }

    fun isInSameSet(a: Int, b: Int): Boolean {
        return find(a) == find(b)
    }

    private fun find(x: Int): Int {
        if (father[x] != x) {
            father[x] = find(father[x]) // Path compression
        }
        return father[x]
    }

    fun union(a: Int, b: Int) {
        val rootA = find(a)
        val rootB = find(b)

        if (rootA != rootB) {
            father[rootA] = rootB
            size[rootB] += size[rootA]
            countSets--
        }
    }

    fun sizeOfSetContains(x: Int): Int {
        return size[find(x)]
    }

    private fun contains(x: Int): Boolean {
        return father[x] != impossibleValue
    }
}
```
