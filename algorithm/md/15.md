```kotlin
class UnionFind(min: Int, max: Int) {

    private val father: IntArray = IntArray(max + 1)
    private val sizes: IntArray = IntArray(max + 1)

    init {
        for (i in min..max) {
            father[i] = i
            sizes[i] = 1
        }
    }

    fun union(a: Int, b: Int) {
        val rootA = find(a)
        val rootB = find(b)
        if (rootA != rootB) {
            father[rootA] = rootB
            sizes[rootB] += sizes[rootA]
        }
    }

    fun isInSameSet(a: Int, b: Int): Boolean {
        return find(a) == find(b)
    }

    fun find(x: Int): Int {
        if (father[x] == x) {
            return x
        }
        father[x] = find(father[x])
        return father[x]
    }

    fun sizeOfSetContains(x: Int): Int {
        return sizes[find(x)]
    }
}
```

```kotlin
/**
 * Definition for a category handler.
 * class CategoryHandler(categories: IntArray) {
 *     fun haveSameCategory(a: Int, b: Int): Boolean
 * }

class Solution {
    fun numberOfCategories(n: Int, categoryHandler: CategoryHandler): Int {
        val uf = UnionFind(n)
        for (i in 0 until n) {
            for (j in i + 1 until n) {
                if (categoryHandler.haveSameCategory(i, j)) {
                    uf.union(i, j)
                }
            }
        }

        return uf.numOfSets
    }

        private val rank = IntArray(n)

        fun union(a: Int, b: Int) {
            val rootA = find(a)
            val rootB = find(b)

                parent[rootA] = rootB

                parent[rootA] = rootB
                rank[rootB]++
            } else {
                parent[rootB] = rootA
            }

            numOfSets--
        }
    }
```

```kotlin
class Solution {
    fun equationsPossible(equations: Array<String>): Boolean {
        val uf = UnionFind(26)
        
        for (eq in equations) {
            if (eq.contains("==")) {
                val first = eq[0] - 'a'
                val second = eq[3] - 'a'
                uf.union(first, second)
            }
        }
        
        for (eq in equations) {
            if (eq.contains("!=")) {
                val first = eq[0] - 'a'
                val second = eq[3] - 'a'
                if (uf.inTheSameSet(first, second)) return false
            }
        }
        
        return true
    }
}
```

UnionFind 第四号模版针对一些union的状态一直在变化，如岛屿II一开始是个空集，然后不停的有新的陆地产生，需要进行union操作

```kotlin
    class UnionFind(private val n: Int) {
        var countSets = 0
        private val parent = IntArray(n) { -1 }
        private val rank = IntArray(n) { 1 }

        fun find(x: Int): Int {
            if (parent[x] != x) {
                parent[x] = find(parent[x]) // Path compression
            }
            return parent[x]
        }

        fun add(newElem: Int) {
            if (!contains(newElem)) {
                parent[newElem] = newElem
                countSets++
            }
        }

        fun contains(x: Int): Boolean {
            return parent[x] != -1
        }

        fun union(a: Int, b: Int) {
            val rootA = find(a)
            val rootB = find(b)
            if (rootA == rootB) return

            // Union by rank
            if (rank[rootA] < rank[rootB]) {
                parent[rootA] = rootB
            } else if (rank[rootA] == rank[rootB]) {
                parent[rootA] = rootB
                rank[rootB]++
            } else {
                parent[rootB] = rootA
            }

            countSets-- // Decrease count of sets as two sets are merged
        }
    }}
```

[Number of Islands II](https://leetcode.com/problems/number-of-islands-ii/description/)
```kotlin
class Solution {
    private val directionX = intArrayOf(0, 0, 1, -1)
    private val directionY = intArrayOf(1, -1, 0, 0)

    fun numIslands2(m: Int, n: Int, positions: Array<IntArray>): List<Int> {
        val res = mutableListOf<Int>()
        val uf = UnionFind(m * n)
        val grid = Array(m) { IntArray(n) { 0 } }

        for (position in positions) {
            val i = position[0]
            val j = position[1]
            val index = i * n + j

            if (grid[i][j] == 1) {
                res.add(uf.countSets)
                continue
            }

            grid[i][j] = 1
            uf.add(index)

            for (k in 0 until 4) {
                val nextX = i + directionX[k]
                val nextY = j + directionY[k]

                if (nextX in 0 until m && nextY in 0 until n && grid[nextX][nextY] == 1) {
                    val neighborIndex = nextX * n + nextY
                    if (uf.contains(neighborIndex)) {
                        uf.union(neighborIndex, index)
                    }
                }
            }

            res.add(uf.countSets)
        }

        return res
    }

    class UnionFind(private val n: Int) {
        var countSets = 0
        private val parent = IntArray(n) { -1 }
        private val rank = IntArray(n) { 1 }

        fun find(x: Int): Int {
            if (parent[x] != x) {
                parent[x] = find(parent[x]) // Path compression
            }
            return parent[x]
        }

        fun add(newElem: Int) {
            if (!contains(newElem)) {
                parent[newElem] = newElem
                countSets++
            }
        }

        fun contains(x: Int): Boolean {
            return parent[x] != -1
        }

        fun union(a: Int, b: Int) {
            val rootA = find(a)
            val rootB = find(b)
            if (rootA == rootB) return

            // Union by rank
            if (rank[rootA] < rank[rootB]) {
                parent[rootA] = rootB
            } else if (rank[rootA] == rank[rootB]) {
                parent[rootA] = rootB
                rank[rootB]++
            } else {
                parent[rootB] = rootA
            }

            countSets-- // Decrease count of sets as two sets are merged
        }
    }
}
```
